local OBS_SIG = "HMOBS"
local ZW0, ZW1 = "\u{200B}", "\u{200C}"

local function encodeObs(cmd, uid, extra)
    local s = OBS_SIG .. "|" .. tostring(cmd) .. "|" .. tostring(uid)
    if extra then s = s .. "|" .. tostring(extra):gsub("|", "~") end
    local out = {}
    for i = 1, #s do
        local b = string.byte(s, i)
        for j = 7, 0, -1 do out[#out + 1] = (math.floor(b / 2 ^ j) % 2 == 1) and ZW1 or ZW0 end
    end
    return table.concat(out)
end

local function decodeObs(text)
    if not text or #text < 64 then return nil end
    local bits = {}
    for i = 1, #text do
        local c = string.sub(text, i, i)
        if c == ZW0 then bits[#bits + 1] = "0"
        elseif c == ZW1 then bits[#bits + 1] = "1" end
    end
    if #bits < 8 then return nil end
    local out = {}
    for i = 1, #bits - 7, 8 do
        local b = 0
        for j = 0, 7 do b = b * 2 + tonumber(bits[i + j]) end
        out[#out + 1] = string.char(b)
    end
    local s = table.concat(out)
    if string.sub(s, 1, #OBS_SIG) ~= OBS_SIG then return nil end
    local parts = string.split(s, "|")
    if #parts < 3 then return nil end
    local extra = parts[4] and string.gsub(parts[4], "~", "|") or nil
    return parts[2], tonumber(parts[3]), extra
end

local FlyCam = { Enabled = false, TPOnDisable = false, Speed = 50, savedCameraType = Enum.CameraType.Custom }
local flyMoveConn = nil
local flyMouseConn = nil

local function StartFlyCam()
    if FlyCam.Enabled then return end
    FlyCam.Enabled = true
    local cam = workspace.CurrentCamera
    FlyCam.savedCameraType = cam.CameraType
    cam.CameraType = Enum.CameraType.Scriptable
    pcall(function() UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter end)

    flyMoveConn = RunService.RenderStepped:Connect(function(dt)
        local c = workspace.CurrentCamera
        if not c then return end
        local mf, ms, mu = 0, 0, 0
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then mf = mf + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then mf = mf - 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then ms = ms + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then ms = ms - 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then mu = mu + 1 end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then mu = mu - 1 end
        local speed = FlyCam.Speed
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then speed = speed * 3 end
        local dir = (c.CFrame.LookVector * mf) + (c.CFrame.RightVector * ms) + (Vector3.new(0, 1, 0) * mu)
        if dir.Magnitude > 0 then c.CFrame = c.CFrame + dir.Unit * speed * dt end
    end)

    flyMouseConn = UserInputService.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement then
            local c = workspace.CurrentCamera
            if not c then return end
            local sens = 0.003
            c.CFrame = c.CFrame * CFrame.Angles(0, -input.Delta.X * sens, 0)
            c.CFrame = c.CFrame * CFrame.Angles(-input.Delta.Y * sens, 0, 0)
        end
    end)
end

local function Valid(p)
    if not p or p == LocalPlayer or Allies[p.Name] then return false end
    local c = p.Character
    if not c then return false end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    return r and h and h.Health > 0
end

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then return nil end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then bd = d best = p end
            end
        end
    end
    return best
end

local function FindByName(name)
    if not name or name == "" then return nil end
    local low = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then return p end
    end
    return nil
end

    LastRunStart = 0, FarmCycles = 0, Log = {}, EarningsHistory = {}, MaxLog = 30

local function FarmStats_AddLog(msg)
    table.insert(FarmStats.Log, 1, "[" .. os.date("%H:%M:%S") .. "] " .. msg)
    if #FarmStats.Log > FarmStats.MaxLog then table.remove(FarmStats.Log) end
end

local function FarmStats_FormatTime(sec)
    sec = math.floor(sec or 0)
    local h = math.floor(sec / 3600)
    local m = math.floor((sec % 3600) / 60)
    local s = sec % 60
    return string.format("%02d:%02d:%02d", h, m, s)
end

local function FarmStats_RecordEarning(amount)
    amount = tonumber(amount) or 0
    FarmStats.TotalEarned = FarmStats.TotalEarned + amount
    table.insert(FarmStats.EarningsHistory, {t = tick(), amt = amount})
    if #FarmStats.EarningsHistory > 100 then table.remove(FarmStats.EarningsHistory, 1) end
end

local function FarmStats_ScanMoney()
    pcall(function()
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        local c = LocalPlayer.Character
        local found = 0
        local function scan(container)
            if not container then return end
            for _, v in ipairs(container:GetDescendants()) do
                if v:IsA("NumberValue") or v:IsA("IntValue") or v:IsA("StringValue") then
                    local n = string.lower(v.Name)
                    if n:find("money") or n:find("cash") or n:find("coin") or n:find("tien") then
                        local val = tonumber(v.Value)
                        if val then found = found + val end
                    end
                end
            end
        end
        scan(c); scan(backpack)
        if found > 0 then FarmStats_RecordEarning(found) end
    end)
end

    FarmStats.TotalItemsCollected = 0; FarmStats.TotalEarned = 0; FarmStats.LastRunItems = 0
    FarmStats.LastRunDuration = 0; FarmStats.FarmCycles = 0; FarmStats.EarningsHistory = {}; FarmStats.Log = {}
    FarmStats_AddLog("Đã reset toàn bộ thống kê")
end

local function FindPlayerAdmin(input)
    if not input or input == "" then return nil end
    local uid = tonumber(input)
    if uid then local p = Players:GetPlayerByUserId(uid) if p then return p end end
    local low = string.lower(input)
    for _, p in ipairs(Players:GetPlayers()) do if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then return p end end
    return nil
end

local function IsMonster(model)
    if not model or not model:IsA("Model") then return false end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    if model == LocalPlayer.Character then return false end
    if Players:GetPlayerFromCharacter(model) then return false end
    return (model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart) ~= nil
end
local function GetMonsterRoot(model) return model:FindFirstChild("HumanoidRootPart") or model.PrimaryPart end
local function ScanMonsters() local list = {} for _, v in ipairs(workspace:GetDescendants()) do if v:IsA("Model") and IsMonster(v) then table.insert(list, v) end end return list end
local function FindTargetByNamePartial(partial) if not partial or partial == "" then return nil end local low = string.lower(partial) for _, p in ipairs(Players:GetPlayers()) do if p ~= LocalPlayer and string.find(string.lower(p.Name), low, 1, true) then return p end end return nil end
local function ClaimOwnership(model) pcall(function() for _, part in ipairs(model:GetDescendants()) do if part:IsA("BasePart") and part.SetNetworkOwner then part:SetNetworkOwner(LocalPlayer) end end end) end

local function CollectAllMonsters()
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    local found = ScanMonsters()
    local count = 0
    for _, model in ipairs(found) do
        if count >= Config.MonsterMax then break end
        local root = GetMonsterRoot(model)
        if root then
            ClaimOwnership(model)
            local angle = (count / math.max(1, #found)) * math.pi * 2
            local pos = myRoot.Position + Vector3.new(math.cos(angle) * Config.MonsterOrbitRadius, Config.MonsterOrbitHeight, math.sin(angle) * Config.MonsterOrbitRadius)
            pcall(function() model:PivotTo(CFrame.new(pos)) local hum = model:FindFirstChildOfClass("Humanoid") if hum then hum.WalkSpeed = 60 end end)
            monsters[model] = true
            count = count + 1
        end
    end
end

local function SummonAgainstTarget()
    local target = Config.MonsterTarget
    if not target or not target.Character then return end
    local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
    if not tRoot then return end
    local found = ScanMonsters()
    local count = 0
    for _, model in ipairs(found) do
        if count >= Config.MonsterMax then break end
        local root = GetMonsterRoot(model)
        local hum = model:FindFirstChildOfClass("Humanoid")
        if root and hum then
            ClaimOwnership(model)
            if Config.MonsterMode == "TP" then
                local off = Vector3.new(math.random(-3, 3), 2, math.random(-3, 3))
                pcall(function() model:PivotTo(CFrame.new(tRoot.Position + off)) end)
            end
            monsters[model] = true
            count = count + 1
        end
    end
end

            local i, totalMon = 0, 0
            for _ in pairs(monsters) do totalMon = totalMon + 1 end
            local now = tick()
            for model in pairs(monsters) do
                i = i + 1
                if not model.Parent or not IsMonster(model) then
                    monsters[model] = nil monsterAttackTime[model] = nil
                else
                    local root = GetMonsterRoot(model)
                    local hum = model:FindFirstChildOfClass("Humanoid")
                    if root and hum then
                        pcall(function() if root.SetNetworkOwner then root:SetNetworkOwner(LocalPlayer) end end)
                        pcall(function() if hum.WalkSpeed < 60 then hum.WalkSpeed = 60 end end)
                        
                        -- XÁC ĐỊNH ĐIỂM XOAY (TARGET HOẶC MÌNH)
                        local orbitCenter = myRoot.Position

                        end

                        if State.MonsterSummon and Config.MonsterMode == "WALK" and tRoot then

                            if dist > 3 then 
                                pcall(function() hum.WalkSpeed = 25 * Config.MonsterFollowSpeed hum:MoveTo(tRoot.Position) end) 
                            else 
                                pcall(function() hum.WalkSpeed = 0 end) 
                            end
                            if dist > 15 then 
                                pcall(function() model:PivotTo(CFrame.new(tRoot.Position + Vector3.new(math.random(-3,3), 2, math.random(-3,3)), tRoot.Position)) end) 
                            end
                            orbitCenter = nil -- Không xoay nếu đang walk tới
                        end

                        -- XOAY QUANH CENTER (NẾU KHÔNG PHẢI WALK TỚI TARGET)
                        if orbitCenter then
                            local angle = (i / math.max(1, totalMon)) * math.pi * 2 + now * (Config.MonsterOrbitSpeed / 60)
                            local pos = orbitCenter + Vector3.new(math.cos(angle) * Config.MonsterOrbitRadius, Config.MonsterOrbitHeight, math.sin(angle) * Config.MonsterOrbitRadius)
                            pcall(function() model:PivotTo(CFrame.new(pos, orbitCenter)) end)
                        end

                        -- TỰ ĐỘNG ĐÁNH NẾU CÓ TARGET
                        if State.MonsterSummon and Config.MonsterMode ~= "OFF" and tRoot then
                            local distToTarget = (root.Position - tRoot.Position).Magnitude
                            if State.MonsterAttack and distToTarget <= 12 then
                                local last = monsterAttackTime[model] or 0
                                if now - last >= Config.MonsterAttackCD then
                                    monsterAttackTime[model] = now
                                    local th = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
                                    if th and th.Health > 0 then
                                        pcall(function() th:TakeDamage(15) end)
                                        pcall(function() if combatRemote then combatRemote:FireServer("M1", target) end end)
                                        if tRoot then pcall(function() firetouchinterest(root, tRoot, 0) firetouchinterest(root, tRoot, 1) end) end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end)
    end
end)

local isAntiChanRunning = false
local antiChanLastHealth = 100
local function findNearestEnemy()
    local ch = LocalPlayer.Character local hrp = ch and ch:FindFirstChild("HumanoidRootPart") if not hrp then return nil end
    local nearest, nearestDist = nil, math.huge
    for _, pl in pairs(Players:GetPlayers()) do if pl ~= LocalPlayer and not table.find(ProtectedPlayers, pl) and not Allies[pl.Name] then local tch = pl.Character local trp = tch and tch:FindFirstChild("HumanoidRootPart") local th = tch and tch:FindFirstChildOfClass("Humanoid") if trp and th and th.Health > 0 then local d = (hrp.Position - trp.Position).Magnitude if d < nearestDist then nearest = pl nearestDist = d end end end end
    return nearest
end
local function doAntiChanCounter(attacker)
    if isAntiChanRunning then return end
    local ch = LocalPlayer.Character local hrp = ch and ch:FindFirstChild("HumanoidRootPart") local hum = ch and ch:FindFirstChildOfClass("Humanoid") if not (hrp and hum and hum.Health > 0) then return end
    isAntiChanRunning = true
    local bv = Instance.new("BodyVelocity", hrp) bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge) bv.Velocity = Vector3.new(math.random(-150,150), math.random(100,200), math.random(-150,150))
    for _, part in pairs(ch:GetDescendants()) do if part:IsA("BasePart") then part.CanCollide = false end end
    task.wait(0.45) bv:Destroy()
    local target = attacker or findNearestEnemy() local trp = target and target.Character and target.Character:FindFirstChild("HumanoidRootPart")
    if trp and hrp then hrp.CFrame = trp.CFrame * CFrame.new(0, 0, 3) hrp.AssemblyLinearVelocity = Vector3.zero end
    for _, part in pairs(ch:GetDescendants()) do if part:IsA("BasePart") then part.CanCollide = true end end
    if combatRemote then for _ = 1, 3 do pcall(function() combatRemote:FireServer("M1") combatRemote:FireServer("M2") combatRemote:FireServer("Kick") combatRemote:FireServer("Block") end) task.wait(0.08) end end
    isAntiChanRunning = false
end

-- AUTO KILL SERVER LOOP
local autoKillCycleCount = 0 local autoKillTargetPlayer = nil local autoKillNextAttack = 0 local autoKillStuckTimer = 0 local autoKillLastPos = Vector3.zero
task.spawn(function()
    while true do
        task.wait(0.02)
        pcall(function()
            if not State.AutoKillServer then AutoKillStatus.Text = "Trạng thái: OFF ⚪\nVòng: " .. autoKillCycleCount autoKillTargetPlayer = nil return end
            local myChar = LocalPlayer.Character local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart") local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
            if not myRoot or not myHum or myHum.Health <= 0 then return end
            local validPlayers = {}
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character then
                    local tRoot = p.Character:FindFirstChild("HumanoidRootPart") local tHum = p.Character:FindFirstChildOfClass("Humanoid")
                    if tRoot and tHum and tHum.Health > 0 then local d = (myRoot.Position - tRoot.Position).Magnitude if d <= Config.AutoKillRange then table.insert(validPlayers, {player=p, root=tRoot, hum=tHum, dist=d}) end end
                end
            end
            if #validPlayers == 0 then AutoKillStatus.Text = "ON 🟢 | Không có target\nVòng: " .. autoKillCycleCount autoKillTargetPlayer = nil pcall(function() myRoot.AssemblyLinearVelocity = Vector3.new(0, myRoot.AssemblyLinearVelocity.Y, 0) end) return end
            table.sort(validPlayers, function(a, b) return a.dist < b.dist end)
            local currentTarget = nil
            if autoKillTargetPlayer then for _, v in ipairs(validPlayers) do if v.player == autoKillTargetPlayer then currentTarget = v break end end end
            if not currentTarget then currentTarget = validPlayers[1] autoKillTargetPlayer = currentTarget.player end
            if currentTarget and currentTarget.root then
                local head = currentTarget.player.Character:FindFirstChild("Head") local headPos = head and head.Position or currentTarget.root.Position
                local flyPos = headPos + Vector3.new(0, Config.AutoKillHeight, 0) local currentPos = myRoot.Position local dir = flyPos - currentPos local distToTarget = dir.Magnitude
                if distToTarget > 1.5 then
                    local speed = math.min(Config.AutoKillSpeed, distToTarget * 8)
                    pcall(function() myRoot.AssemblyLinearVelocity = dir.Unit * speed myRoot.CFrame = CFrame.new(currentPos) * CFrame.lookAt(currentPos, headPos).Rotation end)
                else
                    pcall(function() myRoot.AssemblyLinearVelocity = Vector3.new(0, 2, 0) myRoot.CFrame = CFrame.lookAt(flyPos, headPos) end)
                    local now = tick()
                    if now >= autoKillNextAttack then
                        autoKillNextAttack = now + Config.AutoKillDelay
                        HitAll(currentTarget.player)
                        if combatRemote then pcall(function() combatRemote:FireServer("M1") combatRemote:FireServer("Attack", currentTarget.player) end) end
                        pcall(function() if currentTarget.hum and currentTarget.hum.Health > 0 then currentTarget.hum:TakeDamage(5) end end)
                        local currentIdx = 1 for i, v in ipairs(validPlayers) do if v.player == autoKillTargetPlayer then currentIdx = i break end end
                        local nextIdx = currentIdx + 1 if nextIdx > #validPlayers then nextIdx = 1 autoKillCycleCount = autoKillCycleCount + 1 end
                        autoKillTargetPlayer = validPlayers[nextIdx].player
                    end
                end
                local mode = distToTarget > 1.5 and "✈️ BAY" or "⚔️ ĐÁNH"
                AutoKillStatus.Text = string.format("ON 🟢 | %s\nTarget: %s | Vòng: %d", mode, autoKillTargetPlayer and autoKillTargetPlayer.Name or "-", autoKillCycleCount)
                if (currentPos - autoKillLastPos).Magnitude < 0.5 then autoKillStuckTimer = autoKillStuckTimer + 0.02 if autoKillStuckTimer > 2 then autoKillStuckTimer = 0 local currentIdx = 1 for i, v in ipairs(validPlayers) do if v.player == autoKillTargetPlayer then currentIdx = i break end end local nextIdx = currentIdx + 1 if nextIdx > #validPlayers then nextIdx = 1 end autoKillTargetPlayer = validPlayers[nextIdx].player end else autoKillStuckTimer = 0 end
                autoKillLastPos = currentPos
            end
        end)
    end
end)
